target_include_directories(ufo_streaming_tests PRIVATE Source)

add_test(NAME streaming_cache COMMAND ufo_streaming_tests)

if(APPLE)
  # Production bundles retain CMAKE_OSX_ARCHITECTURES. Test executables and the
  # JUCE module sources they compile need only the architecture running them.
  set(UFO_NATIVE_TEST_TARGETS ufo_engine_tests ufo_sample_tests ufo_streaming_tests
    ufo_stream_asset_tests ufo_plugin_tests ufo_gpu_tests ufo_vst3_host_tests)
  set(UFO_ARCHITECTURE_REPORT "${CMAKE_CURRENT_BINARY_DIR}/ufo-architecture-targets.txt")
  file(WRITE "${UFO_ARCHITECTURE_REPORT}" "Host architecture: ${CMAKE_HOST_SYSTEM_PROCESSOR}\n")
  foreach(target IN LISTS UFO_NATIVE_TEST_TARGETS)
    if(TARGET ${target})
      set_property(TARGET ${target} PROPERTY OSX_ARCHITECTURES "${CMAKE_HOST_SYSTEM_PROCESSOR}")
      get_target_property(architectures ${target} OSX_ARCHITECTURES)
      message(STATUS "Native test ${target}: ${architectures}")
      file(APPEND "${UFO_ARCHITECTURE_REPORT}" "${target}: ${architectures}\n")
    endif()
  endforeach()
  foreach(target UFO UFO_Standalone UFO_AU UFO_VST3)
    if(TARGET ${target})
      get_target_property(architectures ${target} OSX_ARCHITECTURES)
      message(STATUS "Production ${target}: ${architectures}")
      file(APPEND "${UFO_ARCHITECTURE_REPORT}" "${target}: ${architectures}\n")
    endif()
  endforeach()
endif()
